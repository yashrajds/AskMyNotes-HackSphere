# AskMyNotes - Requirements
# =========================
# This file lists all dependencies required to run the project.
# 
# Setup Instructions:
# 1. Install Node.js (v18 or higher recommended)
# 2. Install pnpm (recommended) or npm or yarn
#    - pnpm: npm install -g pnpm
#    - yarn: npm install -g yarn
# 3. Install dependencies: pnpm install (or npm install / yarn install)
# 4. Run development server: pnpm dev (or npm run dev / yarn dev)
# 5. Build for production: pnpm build (or npm run build / yarn build)

# =========================
# Node.js Requirements
# =========================

npm >= 9.0.0 (or pnpm >= 8.0.0 or yarn >= 1.22.0)
